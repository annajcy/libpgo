\section{Problem Setup and Notation}

\begin{frame}{Question, Goal, and Scope}
  \begin{block}{Question}
    Given deformation gradients and target first Piola stresses
    \[
      \mathcal D=\{(F_c,P_c^{\star})\}_{c=1}^{N},
    \]
    can the paper-style Systematic Poking parameters be fitted robustly using
    analytic parameter derivatives?
  \end{block}

  \vspace{0.4em}
  \begin{columns}[T]
    \begin{column}{0.49\linewidth}
      \textbf{Covered here}
      \begin{itemize}
        \item 17 stretch-curvature parameters
        \item One volumetric coefficient $\lambda$
        \item $P(F)$ residual and analytic Jacobian
        \item Positive-parameter Gauss--Newton
        \item Train/holdout evaluation and linear oracle
      \end{itemize}
    \end{column}
    \begin{column}{0.49\linewidth}
      \textbf{Not covered yet}
      \begin{itemize}
        \item Unknown equilibrium state $u^\star(p)$
        \item Poking contact and force curves
        \item Equilibrium sensitivity
        \item Collapse and inversion extension
      \end{itemize}
    \end{column}
  \end{columns}
\end{frame}

\section{Materials and Parameterization}

\begin{frame}{Target: Compressible Neo-Hookean Material}
  \small
  Given Young's modulus $E$ and Poisson ratio $\nu$,
  \[
    \mu=\frac{E}{2(1+\nu)},
    \qquad
    \lambda=\frac{E\nu}{(1+\nu)(1-2\nu)}.
  \]

  For principal stretches $s_1,s_2,s_3>0$ and
  $J=s_1s_2s_3$,
  \[
    \psi_{\mathrm{NH}}
    =
    \sum_{i=1}^{3}
    \left[
      \frac{\mu}{2}(s_i^2-1)-\mu\log s_i
    \right]
    +
    \frac{\lambda}{2}(\log J)^2.
  \]

  Therefore, the one-dimensional Valanis--Landel stretch term satisfies
  \[
    f_{\mathrm{NH}}''(s)
    =
    \mu\left(1+\frac{1}{s^2}\right).
  \]

  \begin{block}{Synthetic ground truth}
    $E^\star=2.0\times10^5,\ \nu^\star=0.35$.
    The initial guess uses $E_0=1.2\times10^5,\ \nu_0=0.25$.
  \end{block}
\end{frame}

\begin{frame}{Fitted Model: Paper-Style Systematic Poking}
  \small
  \[
    \psi_{\mathrm{SP}}(s_1,s_2,s_3;p)
    =
    \sum_{i=1}^{3} f(s_i;q)
    +
    \lambda\,\widehat h(J).
  \]

  Stretch curvature uses piecewise-linear hat functions:
  \[
    f''(s;q)=\sum_{k=0}^{K-1}q_k\phi_k(s),
    \qquad q_k=f''(s_k).
  \]
  The spline is integrated subject to
  \[
    f(1)=0,\qquad f'(1)=0.
  \]

  The volumetric spline shape is fixed by log-squared curvature samples:
  \[
    \widehat h''(J_k)
    =
    \frac{1-\log J_k}{J_k^2},
    \qquad
    \widehat h(1)=\widehat h'(1)=0.
  \]

  \[
    p=
    \begin{bmatrix}
      q_0&\cdots&q_{16}&\lambda
    \end{bmatrix}^{T}\in\mathbb R^{18}.
  \]
\end{frame}

\begin{frame}{Positive-Domain Constraint via Log Parameters}
  \small
  Direct optimization over $p$ can produce negative curvature. Instead, use
  \[
    p_i=\eref\exp(\theta_i),
    \qquad \theta_i\in\mathbb R.
  \]

  The complete optimization trajectory then satisfies
  \[
    q_k>0,\qquad \lambda>0.
  \]

  The initial parameters are sampled from the initial Neo-Hookean material:
  \[
    q_{k,0}
    =
    \mu_0\left(1+\frac{1}{s_k^2}\right),
    \qquad
    \lambda_0
    =
    \frac{E_0\nu_0}{(1+\nu_0)(1-2\nu_0)}.
  \]

  \begin{alertblock}{Numerical role}
    The variables $\theta$ are dimensionless. Scaling by $\eref$ gives
    stress-scale physical parameters, removes inequality constraints, and
    prevents line search from leaving the positive parameter domain.
  \end{alertblock}
\end{frame}

\section{Stress Extraction and Objective}

\begin{frame}{Extracting $P(F)$ I: Affine Kinematics}
  \small
  Let $n_u=3n_v$ be the number of nodal displacement degrees of freedom.
  For vertex $a$ at rest position $X_a$, prescribe
  \[
    u_a(F)=(F-I)X_a,
    \qquad
    u(F)\in\mathbb R^{n_u}.
  \]

  Stack the nine entries of $F$ into $\vect{F}\in\mathbb R^9$ and define
  \[
    \underbrace{B}_{n_u\times9}
    =
    \frac{\partial u(F)}{\partial\vect{F}},
    \qquad
    \underbrace{g(u,p)}_{n_u\times1}
    =
    \frac{\partial E_h(u,p)}{\partial u}.
  \]
  Thus $\delta u=B\,\delta\vect{F}$; $B$ is an affine
  \alert{kinematic map}, not a constitutive matrix.

  \begin{block}{Why no static solve?}
    Once $F$ is prescribed, every nodal displacement $u_a(F)$ is known.
    We evaluate the energy gradient $g$ at this state directly; $u$ is not an
    unknown equilibrium variable.
  \end{block}
\end{frame}

\begin{frame}{Extracting $P(F)$ II: Two Forms of Virtual Work}
  \small
  Let $\Omega_0$ be the reference domain and define its reference volume
  \[
    V_0=\int_{\Omega_0}dV.
  \]
  The FEM chain rule gives
  \[
    \delta E_h
    =
    g^T\delta u
    =
    g^TB\,\delta\vect{F}.
  \]
  Continuum virtual work gives
  \[
    \delta E_h
    =
    \int_{\Omega_0}P:\delta F\,dV.
  \]
  Under the homogeneous affine deformation, $F$, $\delta F$, and $P$ are
  spatially constant. Therefore,
  \[
    \int_{\Omega_0}P:\delta F\,dV
    =
    V_0(P:\delta F)
    =
    V_0\,\vect{P}^{\,T}\delta\vect{F}.
  \]

  Equating the two expressions for arbitrary $\delta F$ yields
  \[
    \boxed{
      \vect{P(F,p)}
      =
      \frac{1}{V_0}B^Tg(u(F),p)
    }.
  \]

  For the unit cube used here, $V_0=1$.
\end{frame}

\begin{frame}{Loss: Normalized Stress Residual}
  \small
  For deformation case $c$,
  \[
    r_c(\theta)
    =
    \frac{
      \vect{P_{\mathrm{SP}}(F_c;p(\theta))
      -
      P_{\mathrm{NH}}(F_c)}
    }{\eref}
    \in\mathbb R^9.
  \]

  Stack all training cases:
  \[
    r(\theta)
    =
    \begin{bmatrix}
      r_1(\theta)\\
      \vdots\\
      r_N(\theta)
    \end{bmatrix},
    \qquad
    \boxed{
      \Phi(\theta)=\frac12\norm{r(\theta)}_2^2
    }.
  \]

  \begin{block}{Why divide by $\eref$?}
    The residual, Jacobian, and objective become dimensionless, preventing the
    $\mathcal O(10^5)$ material scale from degrading numerical conditioning.
  \end{block}
\end{frame}

\section{Analytic Parameter Derivatives}

\begin{frame}{Analytic Parameter Jacobian I: The Mixed Derivative}
  \small
  Let $n_u=3n_v$ and $n_p=18$. With
  \[
    p=
    \begin{bmatrix}q_0&\cdots&q_{16}&\lambda\end{bmatrix}^{T}
    \in\mathbb R^{n_p},
    \qquad
    g(u,p)=\frac{\partial E_h}{\partial u}
    \in\mathbb R^{n_u},
  \]
  where $g$ is the nodal energy-gradient (internal-force) vector, define
  \[
    E_{up}
    =
    \frac{\partial g}{\partial p}
    =
    \frac{\partial^2E_h}{\partial u\,\partial p}
    \in\mathbb R^{n_u\times n_p}.
  \]

  \[
    E_{up}
    =
    \begin{bmatrix}
      \dfrac{\partial g}{\partial q_0}
      &\cdots&
      \dfrac{\partial g}{\partial q_{16}}
      &
      \dfrac{\partial g}{\partial\lambda}
    \end{bmatrix}.
  \]

  \begin{block}{Meaning}
    Holding $u$ fixed, $\delta g=E_{up}\delta p$. Column $k$ is therefore the
    change in the entire nodal internal-force vector caused by a unit change
    of $p_k$. Hence $E_{up}$ is a \emph{material-sensitivity matrix}, not the
    displacement Hessian.
  \end{block}
\end{frame}

\begin{frame}{Analytic Parameter Jacobian II: Stress Sensitivity}
  \small
  Recall the affine kinematic map
  \[
    B=\frac{\partial u(F)}{\partial\vect F}
      \in\mathbb R^{n_u\times9}
  \]
  and the stress extraction formula
  \[
    \vect P=\frac1{V_0}B^Tg.
  \]

  At fixed $F_c$, both $u(F_c)$ and $B$ are independent of $p$. Therefore,
  \[
    \delta\vect P
    =
    \frac1{V_0}B^T\delta g
    =
    \frac1{V_0}B^TE_{up}\,\delta p.
  \]
  Hence
  \[
    \boxed{
      \frac{\partial\vect{P(F_c,p)}}{\partial p}
      =
      \frac1{V_0}
      \underbrace{B^T}_{9\times n_u}
      \underbrace{E_{up}}_{n_u\times n_p}
    }
    \in\mathbb R^{9\times n_p}.
  \]

  \begin{block}{Interpretation}
    $E_{up}$ maps material changes to nodal internal-force changes; $B^T/V_0$
    then maps those nodal changes to first Piola stress changes.
  \end{block}
\end{frame}

\begin{frame}{Analytic Parameter Jacobian III: Log Parameters}
  \small
  Positivity is enforced by
  \[
    p_i=\eref\exp(\theta_i),
    \qquad
    \frac{\partial p_i}{\partial\theta_j}=p_i\delta_{ij},
    \qquad
    \frac{\partial p}{\partial\theta}
    =
    \operatorname{diag}(p).
  \]

  Recall the normalized residual
  \[
    r_c(\theta)
    =
    \frac{
      \vect{P_{\mathrm{SP}}(F_c;p(\theta))-P_{\mathrm{NH}}(F_c)}
    }{\eref}
    \in\mathbb R^9.
  \]
  The chain rule gives
  \[
    \boxed{
      J_c
      :=
      \frac{\partial r_c}{\partial\theta}
      =
      \frac1{V_0\eref}
      \underbrace{B^T}_{9\times n_u}
      \underbrace{E_{up}}_{n_u\times n_p}
      \underbrace{\operatorname{diag}(p)}_{n_p\times n_p}
    }
    \in\mathbb R^{9\times n_p}.
  \]

  Stacking $N$ cases gives $J\in\mathbb R^{9N\times n_p}$.
  Central finite differences give
  \[
    \frac{\norm{J-J_{\mathrm{FD}}}_F}
         {\norm{J_{\mathrm{FD}}}_F}
    =4.8\times10^{-10}.
  \]
\end{frame}

\section{Optimization and Verification}

\begin{frame}{Damped Gauss--Newton with Line Search}
  \small
  At each iteration,
  \[
    g_\theta=J^Tr,
    \qquad
    H_{\mathrm{GN}}=J^TJ,
  \]
  and the step solves
  \[
    \boxed{
      \left[
        J^TJ+\gamma\,\operatorname{diag}(J^TJ)
      \right]\Delta\theta
      =
      -J^Tr
    }.
  \]

  \begin{enumerate}
    \item Restrict $\norm{\Delta\theta}_\infty\leq1$ to avoid exponential jumps.
    \item Backtrack over $\alpha\in\{1,\frac12,\frac14,\ldots\}$.
    \item Accept a step when $\Phi(\theta+\alpha\Delta\theta)<\Phi(\theta)$.
    \item On success, set $\gamma\leftarrow\gamma/3$; on failure,
      set $\gamma\leftarrow10\gamma$.
    \item Stop on gradient tolerance, step tolerance, or failure to decrease.
  \end{enumerate}
\end{frame}

\begin{frame}{Key Diagnosis: Why Was the Initial Design Rank Deficient?}
  \small
  \begin{columns}[T]
    \begin{column}{0.48\linewidth}
      \begin{alertblock}{Initial setup}
        Every observed stretch lay strictly inside $(0.5,2)$. The two endpoint
        hat functions never entered an integration interval, so
        \[
          \operatorname{rank}(A)=16<18.
        \]
        The outer parameters were unidentifiable, and the unconstrained oracle
        contained zero and negative entries.
      \end{alertblock}
    \end{column}
    \begin{column}{0.48\linewidth}
      \begin{block}{Corrected setup}
        Add one training state at every non-rest knot:
        \[
          F_k=\operatorname{diag}(s_k,1,1).
        \]
        The corrected design has
        \[
          \operatorname{rank}(A)=18,
          \qquad
          \kappa(A)=800.3.
        \]
      \end{block}
    \end{column}
  \end{columns}

  \vspace{0.7em}
  \centering
  \textbf{An optimizer cannot recover material directions that the data never excite.}
\end{frame}

\begin{frame}{Why the Model Is Linear in Physical Parameters}
  \small
  The curvature spline is represented by fixed hat functions $\phi_k$:
  \[
    f''(s;q)=\sum_{k=0}^{16}q_k\phi_k(s).
  \]
  With $f(1)=f'(1)=0$, integration gives
  \[
    f(s;q)
    =
    \sum_{k=0}^{16}q_k\Phi_k(s),
    \qquad
    \Phi_k(s)=\int_1^s(s-t)\phi_k(t)\,dt.
  \]
  Integration preserves linearity in $q_k$. The fixed volume shape is scaled
  by one parameter:
  \[
    h(J;\lambda)=\lambda\,\widehat h(J).
  \]

  Thus, for fixed $F$,
  \[
    \boxed{
      \vect{P_{\mathrm{SP}}(F;p)}=A(F)p,
      \qquad
      p=\begin{bmatrix}q_0&\cdots&q_{16}&\lambda\end{bmatrix}^T
    }.
  \]

  \begin{alertblock}{What ``linear'' means here}
    $P$ is linear in $p$, but generally nonlinear in $F$. After
    $p=\eref\exp\theta$, it is also nonlinear in $\theta$.
  \end{alertblock}
\end{frame}

\begin{frame}{Independent Check: Linear Oracle}
  \small
  For each fixed training deformation $F_c$, define
  \[
    A_c=A(F_c)\in\mathbb R^{9\times18},
    \qquad
    b_c=\frac{\vect{P^\star(F_c)}}{\eref},
    \qquad
    \bar p=\frac{p}{\eref}.
  \]

  Stack every case:
  \[
    A=
    \begin{bmatrix}A_1\\ \vdots\\ A_N\end{bmatrix},
    \qquad
    b=
    \begin{bmatrix}b_1\\ \vdots\\ b_N\end{bmatrix},
    \qquad
    A\bar p\approx b.
  \]

  The global unconstrained least-squares solution is obtained directly:
  \[
    \boxed{
      \bar p_{\mathrm{oracle}}
      =
      \arg\min_{\bar p}\norm{A\bar p-b}_2^2
      =
      A^+b
    }.
  \]
\end{frame}

\begin{frame}{Why the Linear Oracle Is an Independent Check}
  \small
  \begin{itemize}
    \item It solves the physical-parameter least-squares problem directly with
      an SVD or QR factorization.
    \item It does not use log-space Gauss--Newton, damping, or line search.
    \item If $\operatorname{rank}(A)=18$, the unconstrained linear solution is
      unique.
    \item If every oracle parameter is positive, it is also feasible under
      $p=\eref\exp\theta$.
  \end{itemize}

  \end{block}

\end{frame}

\begin{frame}{Why No Static Solve? I: The State Is Observed}
  \small
  Each calibration datum already contains a measured deformation gradient:
  \[
    \mathcal D=\{(F_c,P_c^\star)\}_{c=1}^{N}.
  \]
  During material optimization, $F_c$ is fixed data and does not depend on
  $p$:
  \[
    \frac{d\,\vect{F_c}}{dp}=0.
  \]

  The corresponding affine nodal displacement is explicitly known:
  \[
    u_a(F_c)=(F_c-I)X_a.
  \]
  Thus the complete forward evaluation is
  \[
    (F_c,p)
    \longrightarrow
    u(F_c)
    \longrightarrow
    g(u(F_c),p)
    \longrightarrow
    P(F_c,p).
  \]

  \begin{alertblock}{No equilibrium equation is imposed}
    The affine displacement is enforced at all nodes; there are no unknown
    free DOFs to minimize over. In general $g(u(F_c),p)\ne0$: its entries are
    the nodal reactions required to maintain the prescribed homogeneous
    deformation, not an equilibrium error that must be driven to zero.
  \end{alertblock}
\end{frame}

\begin{frame}{Why No Static Solve? II: The Derivative Consequence}
  \small
  The normalized stress residual for datum $c$ is
  \[
    r_c(p)
    =
    \frac{1}{\eref}
    \vect{P(F_c,p)-P_c^\star}.
  \]
  The total parameter derivative is
  \[
    \frac{dr_c}{dp}
    =
    \frac{1}{\eref}
    \left[
      \underbrace{\frac{\partial\vect{P}}{\partial p}}_{\text{material effect}}
      +
      \underbrace{\frac{\partial\vect{P}}{\partial\vect F}
      \frac{d\vect{F_c}}{dp}}_{=\,0\text{ because }F_c\text{ is data}}
    \right]
    =
    \boxed{
      \frac{1}{\eref V_0}B^TE_{up}
    }.
  \]

  Therefore, direct stress fitting requires neither a nonlinear state solve
  nor a stiffness inverse.

  By contrast, a reaction-force experiment has unknown free displacement:
  \[
    g_f(u_f^\star,u_c,p)=0,
    \qquad
    \frac{du_f^\star}{dp}=-K_{ff}^{-1}E_{up,f}.
  \]
  That state-sensitivity term is precisely what the later static-fitting
  experiment adds.
\end{frame}
